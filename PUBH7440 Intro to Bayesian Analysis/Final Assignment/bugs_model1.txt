  for(i in 1:N){
    Y[i] ~ dpois(nlambda[i])
    nlambda[i] <- n[i] * lambda[i]
    lambda[i] <- exp(theta[i])
    theta[i] <- beta0[race[i],sex[i]] + beta1[race[i],sex[i]] * (year[i]-1)
  }
  
  for(r in 1:Nr){    #race
    for(s in 1:Ng){  #sex
      beta0[r,s] ~ dnorm(0,0.001)
      beta1[r,s] ~ dnorm(0,0.001)
    }
  }

for (s in 1:2) {  
	for (t in 1:10) {    
		log_ratio[s, t] <- (beta0[1, s] - beta0[2, s]) + (beta1[1, s] - beta1[2, s]) * (t - 1)    
		ratio_bw[s, t] <- exp(log_ratio[s, t])  
	}
}

  junk <- county[1] + Ns + Ny #gotta use these to avoid errors
